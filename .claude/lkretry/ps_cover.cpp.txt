// capture-lifecycle r1 (PS): a STILL screen for the still-screen run. The worker duplicates this PC's primary
// monitor, and a desktop that animates anything (a clock, a video, a blinking caret) is not the 10-22 logon
// screen. So, for that run only, a static window covers the primary monitor: topmost, no activation, click-
// through, its own message thread (a window that stops answering is ghosted by DWM -- a change), and its text
// says what it is and that it goes away by itself. Under it nothing the user does changes the picture
// except the cursor (a pointer-only update, never content). Removed at the end of the run.
struct StillCover {
  HWND hwnd = nullptr;
  std::thread t;
  std::atomic<bool> up{false};
  DWORD threadId = 0;
  static LRESULT CALLBACK Proc(HWND h, UINT m, WPARAM w, LPARAM l) {
    if (m == WM_PAINT) {
      PAINTSTRUCT ps;
      HDC dc = BeginPaint(h, &ps);
      RECT rc;
      GetClientRect(h, &rc);
      HBRUSH bg = CreateSolidBrush(RGB(40, 64, 96));
      FillRect(dc, &rc, bg);
      DeleteObject(bg);
      SetBkMode(dc, TRANSPARENT);
      SetTextColor(dc, RGB(235, 235, 235));
      HFONT f = CreateFontW(40, 0, 0, 0, FW_SEMIBOLD, 0, 0, 0, DEFAULT_CHARSET, 0, 0, CLEARTYPE_QUALITY, 0, L"Segoe UI");
      HGDIOBJ old = SelectObject(dc, f);
      DrawTextW(dc,
                L"GNLink test: a still screen (capture-lifecycle PS)\n"
                L"This cover goes away by itself in about one minute.",
                -1, &rc, DT_CENTER | DT_VCENTER | DT_WORDBREAK);
      SelectObject(dc, old);
      DeleteObject(f);
      EndPaint(h, &ps);
      return 0;
    }
    if (m == WM_NCHITTEST) return HTTRANSPARENT;
    if (m == WM_MOUSEACTIVATE) return MA_NOACTIVATE;
    return DefWindowProcW(h, m, w, l);
  }
  bool Start() {
    t = std::thread([this] {
      threadId = GetCurrentThreadId();
      WNDCLASSW wc{};
      wc.lpfnWndProc = Proc;
      wc.hInstance = GetModuleHandleW(nullptr);
      wc.lpszClassName = L"GNLinkStillScreenCover";
      wc.hCursor = LoadCursorW(nullptr, reinterpret_cast<LPCWSTR>(IDC_ARROW));
      RegisterClassW(&wc);
      const int w = GetSystemMetrics(SM_CXSCREEN), h = GetSystemMetrics(SM_CYSCREEN);  // the primary monitor
      hwnd = CreateWindowExW(WS_EX_TOPMOST | WS_EX_NOACTIVATE | WS_EX_TOOLWINDOW | WS_EX_LAYERED | WS_EX_TRANSPARENT,
                             wc.lpszClassName, L"GNLink still-screen test", WS_POPUP, 0, 0, w, h, nullptr, nullptr,
                             wc.hInstance, nullptr);
      if (hwnd) {
        SetLayeredWindowAttributes(hwnd, 0, 255, LWA_ALPHA);  // opaque; layered only for the click-through
        ShowWindow(hwnd, SW_SHOWNOACTIVATE);
        UpdateWindow(hwnd);
      }
      up.store(hwnd != nullptr);
      MSG msg;
      while (GetMessageW(&msg, nullptr, 0, 0) > 0) {
        TranslateMessage(&msg);
        DispatchMessageW(&msg);
      }
      if (hwnd) DestroyWindow(hwnd);
    });
    return wait_until([&] { return up.load(); }, 3000);
  }
  void Stop() {
    if (threadId) PostThreadMessageW(threadId, WM_QUIT, 0, 0);
    if (t.joinable()) t.join();
  }
};
